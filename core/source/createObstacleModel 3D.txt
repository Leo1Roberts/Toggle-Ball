// Old 3D model generation code

void Obstacle::createObstacleModel() {
	obstacle.vertices.clear();
	obstacle.indices.clear();
	std::vector<Vertex>& vs = obstacle.vertices;
	std::vector<Index>& is = obstacle.indices;

	float bevel = BEVEL_AMOUNT * getMinorRadius();

	const float HALF_DEPTH = getMinorRadius();


	if (getIsStraight()) {
		// Caps

		vs.emplace_back(vec3(HALF_DEPTH, getStart().y, getStart().z), vec2(), vec3(1, 0, 0), getColor()); // Centre of semicircle
		vs.emplace_back(vec3(HALF_DEPTH, getEnd().y, getEnd().z), vec2(), vec3(1, 0, 0), getColor());
		vs.emplace_back(vec3(-HALF_DEPTH, getStart().y, getStart().z), vec2(), vec3(-1, 0, 0), getColor());
		vs.emplace_back(vec3(-HALF_DEPTH, getEnd().y, getEnd().z), vec2(), vec3(-1, 0, 0), getColor());
		for (int i = 0; i <= SECTORS_PER_SEMICIRCLE; i++) {
			float ang = (float)i / (float)SECTORS_PER_SEMICIRCLE * PI;
			float y = std::sin(ang);
			float z = std::cos(ang);
			float yrv = y * (getMinorRadius() - bevel);
			float zrv = z * (getMinorRadius() - bevel);
			float yrh = y * getMinorRadius();
			float zrh = z * getMinorRadius();
			vs.emplace_back(vec3(HALF_DEPTH, getStart().y + yrv, getStart().z + zrv), vec2(), vec3(1, 0, 0), getColor());
			vs.emplace_back(vec3(HALF_DEPTH - bevel, getStart().y + yrh, getStart().z + zrh), vec2(), vec3(0, y, z), getColor());
			vs.emplace_back(vec3(HALF_DEPTH, getEnd().y - yrv, getEnd().z + zrv), vec2(), vec3(1, 0, 0), getColor());
			vs.emplace_back(vec3(HALF_DEPTH - bevel, getEnd().y - yrh, getEnd().z + zrh), vec2(), vec3(0, -y, z), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH, getStart().y + yrv, getStart().z + zrv), vec2(), vec3(-1, 0, 0), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH + bevel, getStart().y + yrh, getStart().z + zrh), vec2(), vec3(0, y, z), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH, getEnd().y - yrv, getEnd().z + zrv), vec2(), vec3(-1, 0, 0), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH + bevel, getEnd().y - yrh, getEnd().z + zrh), vec2(), vec3(0, -y, z), getColor());
		}
		for (int i = 4; i < SECTORS_PER_SEMICIRCLE * 8 + 3; i += 8) {

			// Semicircles

			is.push_back(i + 8);
			is.push_back(i + 0);
			is.push_back(0);

			is.push_back(i + 2);
			is.push_back(i + 10);
			is.push_back(1);

			is.push_back(i + 4);
			is.push_back(i + 12);
			is.push_back(2);

			is.push_back(i + 14);
			is.push_back(i + 6);
			is.push_back(3);

			// C-shapes

			is.push_back(i + 1);
			is.push_back(i + 9);
			is.push_back(i + 13);
			is.push_back(i + 1);
			is.push_back(i + 13);
			is.push_back(i + 5);

			is.push_back(i + 15);
			is.push_back(i + 11);
			is.push_back(i + 3);
			is.push_back(i + 15);
			is.push_back(i + 3);
			is.push_back(i + 7);

			// Bevel

			is.push_back(i + 0);
			is.push_back(i + 8);
			is.push_back(i + 9);
			is.push_back(i + 0);
			is.push_back(i + 9);
			is.push_back(i + 1);

			is.push_back(i + 2);
			is.push_back(i + 3);
			is.push_back(i + 11);
			is.push_back(i + 2);
			is.push_back(i + 11);
			is.push_back(i + 10);

			is.push_back(i + 13);
			is.push_back(i + 12);
			is.push_back(i + 4);
			is.push_back(i + 5);
			is.push_back(i + 13);
			is.push_back(i + 4);

			is.push_back(i + 15);
			is.push_back(i + 7);
			is.push_back(i + 6);
			is.push_back(i + 14);
			is.push_back(i + 15);
			is.push_back(i + 6);
		}

		// Rectangular faces

		is.push_back(4 + 0); // Front
		is.push_back(4 + 2);
		is.push_back(4 + 2 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 0);
		is.push_back(4 + 2 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 0 + SECTORS_PER_SEMICIRCLE * 8);

		is.push_back(4 + 2); // Bevel
		is.push_back(4 + 0);
		is.push_back(4 + 1);
		is.push_back(4 + 2);
		is.push_back(4 + 1);
		is.push_back(4 + 3);

		is.push_back(4 + 1); // Top
		is.push_back(4 + 5);
		is.push_back(4 + 7);
		is.push_back(4 + 1);
		is.push_back(4 + 7);
		is.push_back(4 + 3);

		is.push_back(4 + 7); // Bevel
		is.push_back(4 + 5);
		is.push_back(4 + 4);
		is.push_back(4 + 7);
		is.push_back(4 + 4);
		is.push_back(4 + 6);

		is.push_back(4 + 4); // Back
		is.push_back(4 + 4 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 6 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 4);
		is.push_back(4 + 6 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 6);

		is.push_back(4 + 6 + SECTORS_PER_SEMICIRCLE * 8); // Bevel
		is.push_back(4 + 4 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 5 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 6 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 5 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 7 + SECTORS_PER_SEMICIRCLE * 8);

		is.push_back(4 + 1 + SECTORS_PER_SEMICIRCLE * 8); // Bottom
		is.push_back(4 + 3 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 7 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 1 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 7 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 5 + SECTORS_PER_SEMICIRCLE * 8);

		is.push_back(4 + 3 + SECTORS_PER_SEMICIRCLE * 8); // Bevel
		is.push_back(4 + 1 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 0 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 3 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 0 + SECTORS_PER_SEMICIRCLE * 8);
		is.push_back(4 + 2 + SECTORS_PER_SEMICIRCLE * 8);
	} else {
		// Caps

		vs.emplace_back(vec3(HALF_DEPTH, getStart().y, getStart().z), vec2(), vec3(1, 0, 0), getColor()); // Centre of semicircle
		vs.emplace_back(vec3(HALF_DEPTH, getEnd().y, getEnd().z), vec2(), vec3(1, 0, 0), getColor());
		vs.emplace_back(vec3(-HALF_DEPTH, getStart().y, getStart().z), vec2(), vec3(-1, 0, 0), getColor());
		vs.emplace_back(vec3(-HALF_DEPTH, getEnd().y, getEnd().z), vec2(), vec3(-1, 0, 0), getColor());
		for (int i = 0; i <= SECTORS_PER_SEMICIRCLE; i++) {
			float ang = (float)i / (float)SECTORS_PER_SEMICIRCLE * PI + getHalfArcAngle();
			float y = std::sin(ang);
			float z = std::cos(ang);
			float yrv = y * (getMinorRadius() - bevel);
			float zrv = z * (getMinorRadius() - bevel);
			float yrh = y * getMinorRadius();
			float zrh = z * getMinorRadius();
			vs.emplace_back(vec3(HALF_DEPTH, getStart().y + yrv, getStart().z + zrv), vec2(), vec3(1, 0, 0), getColor());
			vs.emplace_back(vec3(HALF_DEPTH - bevel, getStart().y + yrh, getStart().z + zrh), vec2(), vec3(0, y, z), getColor());
			vs.emplace_back(vec3(HALF_DEPTH, getEnd().y - yrv, getEnd().z + zrv), vec2(), vec3(1, 0, 0), getColor());
			vs.emplace_back(vec3(HALF_DEPTH - bevel, getEnd().y - yrh, getEnd().z + zrh), vec2(), vec3(0, -y, z), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH, getStart().y + yrv, getStart().z + zrv), vec2(), vec3(-1, 0, 0), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH + bevel, getStart().y + yrh, getStart().z + zrh), vec2(), vec3(0, y, z), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH, getEnd().y - yrv, getEnd().z + zrv), vec2(), vec3(-1, 0, 0), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH + bevel, getEnd().y - yrh, getEnd().z + zrh), vec2(), vec3(0, -y, z), getColor());
		}
		for (int i = 4; i < SECTORS_PER_SEMICIRCLE * 8 + 3; i += 8) {
			is.push_back(i + 8);
			is.push_back(i + 0);
			is.push_back(0);

			is.push_back(i + 2);
			is.push_back(i + 10);
			is.push_back(1);

			is.push_back(i + 4);
			is.push_back(i + 12);
			is.push_back(2);

			is.push_back(i + 14);
			is.push_back(i + 6);
			is.push_back(3);

			is.push_back(i + 1);
			is.push_back(i + 9);
			is.push_back(i + 13);
			is.push_back(i + 1);
			is.push_back(i + 13);
			is.push_back(i + 5);

			is.push_back(i + 15);
			is.push_back(i + 11);
			is.push_back(i + 3);
			is.push_back(i + 15);
			is.push_back(i + 3);
			is.push_back(i + 7);

			// Bevel

			is.push_back(i + 0);
			is.push_back(i + 8);
			is.push_back(i + 9);
			is.push_back(i + 0);
			is.push_back(i + 9);
			is.push_back(i + 1);

			is.push_back(i + 2);
			is.push_back(i + 3);
			is.push_back(i + 11);
			is.push_back(i + 2);
			is.push_back(i + 11);
			is.push_back(i + 10);

			is.push_back(i + 13);
			is.push_back(i + 12);
			is.push_back(i + 4);
			is.push_back(i + 5);
			is.push_back(i + 13);
			is.push_back(i + 4);

			is.push_back(i + 15);
			is.push_back(i + 7);
			is.push_back(i + 6);
			is.push_back(i + 14);
			is.push_back(i + 15);
			is.push_back(i + 6);
		}

		// Banana

		vs.emplace_back(vec3(HALF_DEPTH, 0, 0), vec2(), vec3(1, 0, 0), getColor()); // Centre of arc
		vs.emplace_back(vec3(-HALF_DEPTH, 0, 0), vec2(), vec3(-1, 0, 0), getColor());
		const byte NUM_SECTORS = (byte)std::ceil((float)SECTORS_PER_SEMICIRCLE * 2.0f * getHalfArcAngle() / PI);
		for (int i = 0; i <= NUM_SECTORS; i++) {
			float ang = (float)i / (float)NUM_SECTORS * 2.0f * getHalfArcAngle() - getHalfArcAngle();
			float y = std::sin(ang);
			float z = std::cos(ang);
			float yrov = y * (getMajorRadius() + getMinorRadius() - bevel);
			float zrov = z * (getMajorRadius() + getMinorRadius() - bevel);
			float yriv = y * (getMajorRadius() - getMinorRadius() + bevel);
			float zriv = z * (getMajorRadius() - getMinorRadius() + bevel);
			float yroh = y * (getMajorRadius() + getMinorRadius());
			float zroh = z * (getMajorRadius() + getMinorRadius());
			float yrih = y * (getMajorRadius() - getMinorRadius());
			float zrih = z * (getMajorRadius() - getMinorRadius());
			vs.emplace_back(vec3(HALF_DEPTH, yriv, zriv), vec2(), vec3(1, 0, 0), getColor());
			vs.emplace_back(vec3(HALF_DEPTH - bevel, yrih, zrih), vec2(), vec3(0, -y, -z), getColor());
			vs.emplace_back(vec3(HALF_DEPTH, yrov, zrov), vec2(), vec3(1, 0, 0), getColor());
			vs.emplace_back(vec3(HALF_DEPTH - bevel, yroh, zroh), vec2(), vec3(0, y, z), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH, yriv, zriv), vec2(), vec3(-1, 0, 0), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH + bevel, yrih, zrih), vec2(), vec3(0, -y, -z), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH, yrov, zrov), vec2(), vec3(-1, 0, 0), getColor());
			vs.emplace_back(vec3(-HALF_DEPTH + bevel, yroh, zroh), vec2(), vec3(0, y, z), getColor());
		}
		for (int i = 4 + 8 * (SECTORS_PER_SEMICIRCLE + 1) + 2; i < 4 + 8 * (SECTORS_PER_SEMICIRCLE + 1) + NUM_SECTORS * 8 + 1; i += 8) {
			is.push_back(i + 0);
			is.push_back(i + 8);
			is.push_back(i + 10);
			is.push_back(i + 0);
			is.push_back(i + 10);
			is.push_back(i + 2);

			is.push_back(i + 4);
			is.push_back(i + 6);
			is.push_back(i + 14);
			is.push_back(i + 4);
			is.push_back(i + 14);
			is.push_back(i + 12);

			is.push_back(i + 1);
			is.push_back(i + 5);
			is.push_back(i + 13);
			is.push_back(i + 1);
			is.push_back(i + 13);
			is.push_back(i + 9);

			is.push_back(i + 3);
			is.push_back(i + 11);
			is.push_back(i + 15);
			is.push_back(i + 3);
			is.push_back(i + 15);
			is.push_back(i + 7);

			// Bevel

			is.push_back(i + 9);
			is.push_back(i + 8);
			is.push_back(i + 0);
			is.push_back(i + 1);
			is.push_back(i + 9);
			is.push_back(i + 0);

			is.push_back(i + 11);
			is.push_back(i + 3);
			is.push_back(i + 2);
			is.push_back(i + 10);
			is.push_back(i + 11);
			is.push_back(i + 2);

			is.push_back(i + 4);
			is.push_back(i + 12);
			is.push_back(i + 13);
			is.push_back(i + 4);
			is.push_back(i + 13);
			is.push_back(i + 5);

			is.push_back(i + 6);
			is.push_back(i + 7);
			is.push_back(i + 15);
			is.push_back(i + 6);
			is.push_back(i + 15);
			is.push_back(i + 14);
		}
	}

	obstacle.sendToGpu();
	objShader->setupVertexAttribs();
}
